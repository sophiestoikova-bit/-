function renderNav(){
  nav.innerHTML = "";
  let last = "";
  lessons.forEach((l,i)=>{
    if(l.section !== last){
      const label=document.createElement("div");
      label.className="section";
      label.textContent=l.section;
      nav.appendChild(label);
      last=l.section;
    }
    const b=document.createElement("button");
    b.className="lessonlink"+(i===current?" active":"")+(completed.has(i)?" done":"");
    b.innerHTML=`<span class="dot">${completed.has(i)?"✓":""}</span><span>${i+1}. ${l.title}</span>`;
    b.onclick=()=>showLesson(i);
    nav.appendChild(b);
  });
  const percent=Math.round(completed.size/lessons.length*100);
  document.getElementById("progressText").textContent=percent+"%";
  document.getElementById("progressBar").style.width=percent+"%";
}
function showLesson(i){
  current=Math.max(0,Math.min(lessons.length-1,i));

  document.getElementById("crumb").textContent=l.section;

async function getPyodide(){
  if(!pyodidePromise){
    pyodidePromise=loadPyodide().then(async py=>{
      await py.runPythonAsync(`
import sys, io, contextlib, traceback
def __pythonlab_run(code, test_code):
    ns = {"__builtins__": __builtins__}
    buffer = io.StringIO()
    try:
        with contextlib.redirect_stdout(buffer):
            exec(compile(code, "<урок>", "exec"), ns)
            ns["__out__"] = buffer.getvalue()
            exec(compile(test_code, "<проверка>", "exec"), ns)
        return ("OK", buffer.getvalue())
    except Exception:
        return ("ERROR", buffer.getvalue() + traceback.format_exc())
`);
      document.getElementById("runtime").textContent="Python готов";
      return py;
    }).catch(err=>{
      document.getElementById("runtime").textContent="Не удалось загрузить Python";
      throw err;
    });
  }
  return pyodidePromise;
}
async function runCode(){
  const btn=document.getElementById("runBtn");
  const out=document.getElementById("output");
  const status=document.getElementById("status");
  btn.disabled=true;btn.textContent="Запускаю…";
  out.textContent="Инициализация Python в браузере…";
  status.textContent="";
  try{
    const py=await getPyodide();
    py.globals.set("__USER_CODE",codeBox.value);
    py.globals.set("__TEST_CODE",lessons[current].test);
    const result=py.runPython(`__pythonlab_run(__USER_CODE, __TEST_CODE)`);
    const state=result.get(0), text=result.get(1);
    result.destroy();
    out.textContent=text || (state==="OK"?"Код выполнен без вывода.":"");
    if(state==="OK"){
      completed.add(current);
      localStorage.setItem("pythonlab_completed",JSON.stringify([...completed]));
      status.textContent="✓ Отлично! Задание пройдено.";
      status.className="status ok";
      renderNav();
    }else{
      status.textContent="Пока не пройдено — проверь код и сообщение об ошибке.";
      status.className="status bad";
    }
  }catch(err){
    out.textContent="Не получилось запустить Python.\nПроверь подключение к интернету и перезагрузи страницу.\n\n"+err;
    status.textContent="Среда выполнения недоступна.";
    status.className="status bad";
  }finally{
    btn.disabled=false;btn.textContent="▶ Запустить и проверить";
  }
}
