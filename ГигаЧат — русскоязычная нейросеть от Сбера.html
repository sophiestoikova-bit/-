Вопрос

создай аналог freeCodeCamp с полным курсом по python весь проект помести в один файл index.html

Ответ

Ниже представлен полный код файла index.html. Это самодостаточный одностраничный аналог обучающей платформы.

Проект включает:

  • Адаптивную верстку: боковая панель с навигацией по курсу и основная область с контентом.
  • Интерактивный редактор кода: поле для ввода Python-кода, кнопка запуска и консоль для вывода результата. Выполнение кода происходит на клиентской стороне с помощью библиотеки Brython (Python для браузера).
  • 10 последовательных уроков: от вывода текста и переменных до функций, списков, словарей и базового ООП.
  • Систему отслеживания прогресса: кнопка «Урок пройден» сохраняет номер урока в localStorage, а пройденные уроки отмечаются галочкой в меню.

Чтобы запустить курс, скопируйте весь код ниже, сохраните его в файл с именем index.html и откройте в любом современном браузере.

        function loadLesson(index) {
            if (index < 0 || index >= lessons.length) return;
            currentLesson = index;
            
            const lesson = lessons[index];
            viewer.innerHTML = `
                <h1>${lesson.title}</h1>
                ${lesson.content}
            `;
            codeInput.value = lesson.initialCode;
            output.innerHTML = '';
            
            // Управление кнопками
            prevBtn.disabled = index === 0;
            nextBtn.disabled = index === lessons.length - 1;
            
            // Обновление сайдбара
            const items = lessonList.querySelectorAll('li');
            items.forEach(item => {
                item.classList.remove('active');
                const i = parseInt(item.dataset.lesson);
                if (completedLessons.has(i)) {
                    item.classList.add('completed');
                }
            });
            items[index].classList.add('active');
            
            // Кнопка "Урок пройден"
            if (completedLessons.has(index)) {
                completeBtn.textContent = '✅ Пройдено';
                completeBtn.disabled = true;
            } else {
                completeBtn.textContent = '✅ Урок пройден';
                completeBtn.disabled = false;
            }
        }

        function changeLesson(delta) {
            const newIndex = currentLesson + delta;
            if (newIndex >= 0 && newIndex < lessons.length) {
                loadLesson(newIndex);
            }
        }

        function completeLesson() {
            completedLessons.add(currentLesson);
            saveProgress();
            loadLesson(currentLesson); // Перезагружаем для обновления UI
            if (currentLesson < lessons.length - 1) {
                changeLesson(1);
            }
        }

        function runCode() {
            output.innerHTML = '<em>Выполнение...</em>';
            // Очищаем глобальное пространство Brython от предыдущего кода
            if (window.__BRYTHON__ && window.__BRYTHON__.__dict__) {
                 Object.keys(window.__BRYTHON__.__dict__).forEach(key => {
                    if (!key.startsWith('__') && key !== 'sys' && key !== 'math' && key !== 'builtins') {
                        try { delete window.__BRYTHON__.__dict__[key]; } catch(e) {}
                    }
                });
            }
            try {
                // Выполняем код
                __BRYTHON__.python_to_js(codeInput.value);
                __BRYTHON__.exec($$BRYTHON__.python_to_js(codeInput.value), __BRYTHON__.__globals__);
                // Brython выводит в системную консоль, перенаправим ее в наш div
                const oldLog = console.log;
                console.log = function(message) {
                    oldLog.apply(console, arguments);
                    output.innerHTML += message + '<br>';
                };
                // Принудительно запускаем цикл обработки вывода, если он не сработал
                __BRYTHON__.brython();
            } catch (err) {
                output.innerHTML = `<span style="color: #f07178;">Ошибка: ${err.toString()}</span>`;
            }
        }

        function saveProgress() {
            localStorage.setItem('pythonCourseProgress', JSON.stringify([...completedLessons]));
        }

        function loadProgress() {
            const data = localStorage.getItem('pythonCourseProgress');
            if (data) {
                completedLessons.clear();
                JSON.parse(data).forEach(id => completedLessons.add(id));
            }
        }

        // Кликабельные пункты меню
        lessonList.addEventListener('click', (e) => {
            if (e.target.dataset.lesson) {
                loadLesson(parseInt(e.target.dataset.lesson));
            }
        });

Как пользоваться и дорабатывать проект:

  1. Добавление уроков: Чтобы добавить новый урок, просто добавьте новый объект в массив lessons в JavaScript-коде. Скопируйте структуру существующего объекта, заполните title, content (можно использовать HTML-теги), initialCode (код, который будет в редакторе по умолчанию) и, при необходимости, измените логику кнопок «Вперед»/«Назад».
  2. Сохранение прогресса: Прогресс хранится в localStorage браузера. Если вы очистите кеш или данные сайта, прогресс сбросится. Для серверного хранения потребуется заменить функции saveProgress и loadProgress на отправку запросов к вашему бэкенду.
  3. Ограничения Brython: Код выполняется прямо в браузере пользователя. Это безопасно, но имеет ограничения: вы не сможете использовать библиотеки, требующие системного доступа (например, requests для HTTP-запросов, tkinter для GUI или работу с файлами на диске). Для полноценного бэкенд-курса потребуется интеграция с сервером (например, через API, запускающее код в изолированном контейнере).
